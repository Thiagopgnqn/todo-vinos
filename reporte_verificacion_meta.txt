-## 🔒 Seguridad
+## 🗄️ Modelo de Base de Datos (Prisma)
+
+El esquema de datos (`backend/prisma/schema.prisma`) incluye los siguientes modelos:
 
-- Contraseñas hasheadas con **bcrypt** (12 salt rounds)
-- Autenticación con **JWT** (access token 15min + refresh token 7 días)
-- **Rate limiting** en login y registro (5 intentos/minuto)
-- Validación de datos con **Zod** en backend
-- **Helmet** para headers de seguridad
-- **CORS** configurado
-- Queries parametrizadas vía **Prisma** (previene SQL injection)
-- React escapa HTML por defecto (previene XSS)
-- Variables sensibles en `.env` (nunca hardcodeadas)
+- **`User`**: Cuentas de clientes y administradores con autenticación por hash.
+- **`Product`**: Etiquetas de vino con precio de lista, precio con descuento por transferencia (`transferPrice`), stock disponible, notas de cata, maridaje, bodega, región y contador de ventas (`soldCount`).
+- **`WineType`**: Categorías dinámicas de vinos (Tinto, Blanco, Rosado, Espumante, etc.).
+- **`Order`**: Pedidos registrados con método de entrega (`PICKUP` o `DELIVERY`), datos del cliente, estado y total.
+- **`OrderItem`**: Detalle de productos por pedido con precio unitario y subtotal congelados al momento de compra.
+- **`RefreshToken`**: Tokens de renovación con fecha de expiración y revocación para control de sesiones seguras.

-## 📝 Notas para Producción
+## 📣 Meta Ads (Píxel + API de Conversiones)
+
+Los eventos se envían **por duplicado**: desde el navegador (Píxel, `frontend/src/lib/meta.js`) y desde el servidor (API de Conversiones, vía `POST /api/meta/events` → `backend/src/services/meta.service.js`), **con el mismo `event_id`** para que Meta los deduplique. Los datos personales (email, teléfono, nombre) se envían hasheados con SHA-256 desde el backend. Nada se mide en las rutas `/admin`, y un fallo en el tracking nunca afecta la compra.
+
+| Evento | Cuándo se dispara | Dónde |
+|--------|-------------------|-------|
+| `PageView` | Cada cambio de ruta pública (solo navegador) | `MetaPixelTracker.jsx` |
+| `ViewContent` | Al cargar la ficha de un producto | `ProductPage.jsx` |
+| `AddToCart` | Al agregar unidades efectivamente al carrito (respeta stock) | `CartContext.jsx` |
+| `InitiateCheckout` | Una vez al entrar al checkout con carrito válido | `CheckoutPage.jsx` |
+| `Purchase` | Al crearse el pedido, antes de abrir WhatsApp. `event_id = order-<id>` | `CheckoutForm.jsx` |
+| `Contact` | Soportado por el endpoint; hoy no hay botones de WhatsApp fuera del checkout | — |
+
+**Cómo probar:**
+1. En el Administrador de eventos de Meta → tu Píxel → **Probar eventos**, copiá el código de prueba (ej. `TEST12345`).
+2. Cargalo en `META_TEST_EVENT_CODE` en el backend y reiniciá/redeployá.
+3. Navegá el sitio (con la extensión *Meta Pixel Helper* podés ver los eventos del navegador). En "Probar eventos" deberían aparecer los eventos de **Navegador** y **Servidor**, marcados como deduplicados.
+4. Al terminar, **eliminá `META_TEST_EVENT_CODE`** y redeployá.

+import { Router } from 'express';
+import { z } from 'zod';
+import { sendMetaEvent } from '../services/meta.service.js';
+import { metaLimiter } from '../middleware/rateLimiter.js';
+
+const router = Router();
+
+const schema = z.object({
+  eventName: z.enum([
+    'ViewContent',
+    'AddToCart',
+    'InitiateCheckout',
+    'Purchase',
+    'Contact',
+    'Lead',
+    'Search',
+  ]),
+  eventId: z.string().min(1).max(100),
+  url: z.string().max(2000).optional(),
+  data: z.record(z.string(), z.any()).optional(),
+  user: z
+    .object({
+      email: z.string().max(200).optional(),
+      phone: z.string().max(50).optional(),
+      firstName: z.string().max(100).optional(),
+      lastName: z.string().max(100).optional(),
+    })
+    .optional(),
+  fbp: z.string().max(200).optional(),
+  fbc: z.string().max(300).optional(),
+});
+
+router.post('/events', metaLimiter, (req, res) => {
+  const parsed = schema.safeParse(req.body);
+  if (!parsed.success) {
+    return res.status(400).json({ ok: false });
+  }
+
+  // Respondemos rápido; el envío a Meta continúa en segundo plano
+  res.status(202).json({ ok: true });
+
+  sendMetaEvent({
+    ...parsed.data,
+    ip: req.ip,
+    userAgent: req.headers['user-agent'],
+  }).catch((err) => console.error('[Meta CAPI] Fallo de red', err.message));
+});
+
+export default router;

+import crypto from 'crypto';
+import { env } from '../config/env.js'; // asegura que dotenv ya cargó las variables
+
+const sha256 = (value) =>
+  crypto.createHash('sha256').update(String(value)).digest('hex');
+
+const clean = (v) => (v ? String(v).trim().toLowerCase() : null);
+const cleanPhone = (v) => (v ? String(v).replace(/\D/g, '') : null);
+
+function buildUserData({ user = {}, ip, userAgent, fbp, fbc }) {
+  const ud = {};
+  const em = clean(user.email);
+  const ph = cleanPhone(user.phone);
+  const fn = clean(user.firstName);
+  const ln = clean(user.lastName);
+
+  if (em) ud.em = [sha256(em)];
+  if (ph) ud.ph = [sha256(ph)];
+  if (fn) ud.fn = [sha256(fn)];
+  if (ln) ud.ln = [sha256(ln)];
+  ud.country = [sha256('ar')];
+  if (ip) ud.client_ip_address = ip;
+  if (userAgent) ud.client_user_agent = userAgent;
+  if (fbp) ud.fbp = fbp;
+  if (fbc) ud.fbc = fbc;
+  return ud;
+}
+

### backend/src/services/meta.service.js
import crypto from 'crypto';
import { env } from '../config/env.js'; // asegura que dotenv ya cargó las variables

const sha256 = (value) =>
  crypto.createHash('sha256').update(String(value)).digest('hex');

const clean = (v) => (v ? String(v).trim().toLowerCase() : null);
const cleanPhone = (v) => (v ? String(v).replace(/\D/g, '') : null);

function buildUserData({ user = {}, ip, userAgent, fbp, fbc }) {
  const ud = {};
  const em = clean(user.email);
  const ph = cleanPhone(user.phone);
  const fn = clean(user.firstName);
  const ln = clean(user.lastName);

  if (em) ud.em = [sha256(em)];
  if (ph) ud.ph = [sha256(ph)];
  if (fn) ud.fn = [sha256(fn)];
  if (ln) ud.ln = [sha256(ln)];
  ud.country = [sha256('ar')];
  if (ip) ud.client_ip_address = ip;
  if (userAgent) ud.client_user_agent = userAgent;
  if (fbp) ud.fbp = fbp;
  if (fbc) ud.fbc = fbc;
  return ud;
}

export async function sendMetaEvent({
  eventName,
  eventId,
  url,
  data = {},
  user,
  ip,
  userAgent,
  fbp,

  const pixelId = process.env.META_PIXEL_ID || env.META_PIXEL_ID;
  const token = process.env.META_ACCESS_TOKEN || env.META_ACCESS_TOKEN;
  const version = process.env.META_API_VERSION || env.META_API_VERSION || 'v23.0';
  const testCode = process.env.META_TEST_EVENT_CODE || env.META_TEST_EVENT_CODE;

  if (!pixelId || !token) {
    console.warn('[Meta CAPI] Falta META_PIXEL_ID o META_ACCESS_TOKEN, evento omitido');

  }

  const payload = {
    data: [
      {
        event_name: eventName,
        event_time: Math.floor(Date.now() / 1000),
        event_id: eventId,
        action_source: 'website',
        event_source_url: url,
        user_data: buildUserData({ user, ip, userAgent, fbp, fbc }),
        custom_data: data,
      },
    ],
  };

  // Solo para probar en "Probar eventos" de Meta. Se elimina de las variables al terminar de probar.
  if (testCode) payload.test_event_code = testCode;

  const res = await fetch(
    `https://graph.facebook.com/${version}/${pixelId}/events?access_token=${encodeURIComponent(token)}`,

import { Router } from 'express';
import { z } from 'zod';
import { sendMetaEvent } from '../services/meta.service.js';
import { metaLimiter } from '../middleware/rateLimiter.js';

const router = Router();

const schema = z.object({
  eventName: z.enum([
    'ViewContent',
    'AddToCart',
    'InitiateCheckout',
    'Purchase',
    'Contact',
    'Lead',
    'Search',
  ]),
  eventId: z.string().min(1).max(100),
  url: z.string().max(2000).optional(),
  data: z.record(z.string(), z.any()).optional(),
  user: z
    .object({
      email: z.string().max(200).optional(),
      phone: z.string().max(50).optional(),
      firstName: z.string().max(100).optional(),
      lastName: z.string().max(100).optional(),
    })
    .optional(),
  fbp: z.string().max(200).optional(),
  fbc: z.string().max(300).optional(),
});

router.post('/events', metaLimiter, (req, res) => {
  const parsed = schema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ ok: false });
  }

  // Respondemos rápido; el envío a Meta continúa en segundo plano
  res.status(202).json({ ok: true });

  sendMetaEvent({
    ...parsed.data,
    ip: req.ip,
    userAgent: req.headers['user-agent'],
  }).catch((err) => console.error('[Meta CAPI] Fallo de red', err.message));
});

export default router;

        fbc: getCookie('_fbc'),

}

/**
 * Arma content_ids / contents / num_items a partir de los ítems del carrito.
 * Forma real de los ítems en CartContext: { product, quantity }.
 * Se usa product.price, que es el precio con el que el backend calcula el total del pedido.
 */
export function buildContents(items = []) {
  const contents = (items || []).map((i) => {
    const p = i.product || i;
    return {
      id: String(p.id || p._id),
      quantity: Number(i.quantity) || 1,
      item_price: Number(p.price) || 0,
    };
  });

  return {
    content_type: 'product',
    content_ids: contents.map((c) => c.id),
    contents,
    num_items: contents.reduce((sum, c) => sum + c.quantity, 0),
  };
}
